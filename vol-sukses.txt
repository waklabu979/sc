function scanDirectories($basePath, $maxDepth = 4) {
    $dirs = [];
    scanRecursive($basePath, $dirs, 0, $maxDepth);
    return $dirs;
}

function scanRecursive($dir, &$result, $depth, $maxDepth) {
    if ($depth > $maxDepth) return;
    
    $items = scandir($dir);
    foreach ($items as $item) {
        if ($item == '.' || $item == '..' || $item == 'backup_htaccess') continue;
        $path = $dir . '/' . $item;
        if (is_dir($path)) {
            $result[] = $path;
            scanRecursive($path, $result, $depth + 1, $maxDepth);
        }
    }
}

function deployHtaccess($source, $targets, $backupDir, $chmod = 0444) {
    if (!file_exists($source)) {
        return ['success' => false, 'message' => 'Source .htaccess not found'];
    }
    
    $content = file_get_contents($source);
    $results = [];
    
    if (!is_dir($backupDir)) {
        mkdir($backupDir, 0755, true);
    }
    
    foreach ($targets as $target) {
        $targetFile = $target . '/.htaccess';
        
        try {
            if (file_exists($targetFile)) {
                $backupFile = $backupDir . '/' . 
                             basename($target) . '_' . 
                             date('Ymd_His') . '.bak';
                copy($targetFile, $backupFile);
            }
            
            if (file_put_contents($targetFile, $content)) {
                chmod($targetFile, $chmod);
                $results[$target] = [
                    'status' => 'success', 
                    'message' => '✅ Deployed (chmod ' . decoct($chmod) . ')'
                ];
            } else {
                $results[$target] = [
                    'status' => 'error', 
                    'message' => '❌ Failed to write'
                ];
            }
        } catch (Exception $e) {
            $results[$target] = [
                'status' => 'error', 
                'message' => '❌ ' . $e->getMessage()
            ];
        }
    }
    
    return ['success' => true, 'results' => $results];
}

function deleteHtaccess($targets, $backupDir) {
    $results = [];
    
    if (!is_dir($backupDir)) {
        mkdir($backupDir, 0755, true);
    }
    
    foreach ($targets as $target) {
        $targetFile = $target . '/.htaccess';
        
        try {
            if (!file_exists($targetFile)) {
                $results[$target] = [
                    'status' => 'warning', 
                    'message' => '⚠️ No .htaccess found'
                ];
                continue;
            }
            
            $backupFile = $backupDir . '/' . 
                         basename($target) . '_deleted_' . 
                         date('Ymd_His') . '.bak';
            copy($targetFile, $backupFile);
            
            if (unlink($targetFile)) {
                $results[$target] = [
                    'status' => 'success', 
                    'message' => '✅ Deleted (backup created)'
                ];
            } else {
                $results[$target] = [
                    'status' => 'error', 
                    'message' => '❌ Failed to delete'
                ];
            }
        } catch (Exception $e) {
            $results[$target] = [
                'status' => 'error', 
                'message' => '❌ ' . $e->getMessage()
            ];
        }
    }
    
    return ['success' => true, 'results' => $results];
}

function chmodHtaccess($targets, $mode) {
    $results = [];
    
    foreach ($targets as $target) {
        $targetFile = $target . '/.htaccess';
        
        try {
            if (!file_exists($targetFile)) {
                $results[$target] = [
                    'status' => 'warning', 
                    'message' => '⚠️ No .htaccess found'
                ];
                continue;
            }
            
            if (chmod($targetFile, $mode)) {
                $results[$target] = [
                    'status' => 'success', 
                    'message' => '✅ Chmod changed to ' . decoct($mode)
                ];
            } else {
                $results[$target] = [
                    'status' => 'error', 
                    'message' => '❌ Failed to change chmod'
                ];
            }
        } catch (Exception $e) {
            $results[$target] = [
                'status' => 'error', 
                'message' => '❌ ' . $e->getMessage()
            ];
        }
    }
    
    return ['success' => true, 'results' => $results];
}

        for (var i = 0; i < items.length; i++) {
            var badge = items[i].querySelector('.badge');
            var checkbox = items[i].querySelector('input[type="checkbox"]');
            if (badge && badge.textContent.indexOf('New') !== -1) {
                checkbox.checked = true;
            } else {
                checkbox.checked = false;
            }
        }
        updateSelectedCount();
        showNotification('🆕 Selected directories without .htaccess', 'info');
    }
    
    // Set action dan submit form
    function setAction(action) {
        var checkboxes = document.querySelectorAll('input[name="targets[]"]:checked');
        if (checkboxes.length === 0) {
            showNotification('⚠️ Please select at least one directory', 'warning');
            return;
        }
        
        var actionNames = {
            'deploy': 'Deploy .htaccess',
            'delete': 'Delete .htaccess',
            'chmod': 'Change Chmod'
        };
        
        var confirmMessage = '⚠️ WARNING!\n\n';
        confirmMessage += 'You are about to ' + action + ' .htaccess in ' + checkboxes.length + ' directories.\n\n';
        
        if (action === 'delete') {
            confirmMessage += '⚠️ This will DELETE .htaccess files!\n';
            confirmMessage += 'Files will be backed up before deletion.\n\n';
        } else if (action === 'chmod') {
            var chmodValue = document.getElementById('chmodSelect').value;
            confirmMessage += 'Chmod will be changed to: ' + chmodValue + '\n\n';
        } else {
            var chmodValue = document.getElementById('chmodSelect').value;
            confirmMessage += 'Files will be deployed with chmod: ' + chmodValue + '\n\n';
        }
        
        confirmMessage += 'Are you sure you want to continue?';
        
        if (confirm(confirmMessage)) {
            document.getElementById('actionInput').value = action;
            document.getElementById('actionForm').submit();
        }
    }
    
    // Show notification
    function showNotification(message, type) {
        // Remove existing notification
        var existing = document.querySelector('.custom-notification');
        if (existing) {
            existing.remove();
        }
        
        // Create notification
        var notification = document.createElement('div');
        notification.className = 'custom-notification message ' + type;
        notification.style.cssText = 
            'position: fixed; ' +
            'top: 20px; ' +
            'right: 20px; ' +
            'max-width: 400px; ' +
            'z-index: 9999; ' +
            'animation: slideIn 0.5s ease; ' +
            'box-shadow: 0 4px 20px rgba(0,0,0,0.15); ' +
            'padding: 15px 20px; ' +
            'border-radius: 8px;';
        notification.textContent = message;
        
        document.body.appendChild(notification);
        
        // Auto remove after 3 seconds
        setTimeout(function() {
            notification.style.animation = 'slideOut 0.5s ease';
            setTimeout(function() {
                notification.remove();
            }, 500);
        }, 3000);
    }
